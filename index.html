use obsidian::{Agent, ZkProof, BtcAnchor};

  let agent = Agent::new("my-private-agent")
    .with_model("llama-70b")
    .with_privacy(ZkProof::default())

  agent.deploy().await;
  // Your agent is now live, private, and Bitcoin-secured
}
